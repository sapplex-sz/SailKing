cmake_minimum_required(VERSION 3.24)
project(SailKingWindows VERSION 0.4.0 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
option(SAILKING_BROKER "Build the x64 Rime/translation broker" ON)
option(SAILKING_AVX2_RUNTIME "Build a separate AVX2 translation runtime" OFF)
set(SAILKING_RIME_ROOT "" CACHE PATH "Verified librime distribution root")
set(SAILKING_LLAMA_ROOT "" CACHE PATH "Verified pinned llama.cpp source root")
add_compile_definitions(UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
add_compile_options(/utf-8 /W4 /EHsc /permissive-)
add_library(SailKingTip SHARED Native/Tip.cpp Native/Tip.def Native/Tip.rc)
target_link_libraries(SailKingTip PRIVATE ole32 oleaut32 advapi32 shell32 user32 gdi32 uuid)
add_executable(SailKingProbe Native/Probe.cpp)
target_link_libraries(SailKingProbe PRIVATE ole32 advapi32 shell32 uuid)
add_executable(SailKingCoreTests Tests/CoreTests.cpp)
target_include_directories(SailKingCoreTests PRIVATE Native)
enable_testing()
add_test(NAME Core COMMAND SailKingCoreTests)
if(SAILKING_BROKER)
 if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The broker requires x64; 32-bit TIPs communicate with that broker")
 endif()
 set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
 set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
 set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
 set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
 set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
 set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
 set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
 # Ship both variants; only the isolated broker chooses an optimized DLL.
 foreach(feature SSE42 AVX AVX2 BMI2 FMA F16C)
  set(GGML_${feature} ${SAILKING_AVX2_RUNTIME} CACHE BOOL "" FORCE)
 endforeach()
 set(GGML_AVX512 OFF CACHE BOOL "" FORCE)
 set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
 set(GGML_BLAS OFF CACHE BOOL "" FORCE)
 set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
 set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
 add_subdirectory("${SAILKING_LLAMA_ROOT}" llama)
 add_library(CHaHaRuntime SHARED ../Runtime/src/CHaHaRuntime.cpp)
 target_include_directories(CHaHaRuntime PUBLIC ../Runtime/include)
 target_link_libraries(CHaHaRuntime PRIVATE llama)
 set_target_properties(CHaHaRuntime PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
 if(SAILKING_AVX2_RUNTIME)
  target_compile_definitions(CHaHaRuntime PRIVATE SAILKING_RUNTIME_AVX2)
 endif()
 add_executable(SailKingBroker Native/Broker.cpp Native/RuntimeDispatch.cpp)
 target_link_options(SailKingBroker PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:wmainCRTStartup)
 target_include_directories(SailKingBroker PRIVATE "${SAILKING_RIME_ROOT}/include" ../Runtime/include)
 target_link_libraries(SailKingBroker PRIVATE "${SAILKING_RIME_ROOT}/lib/rime.lib" advapi32 shell32 ole32 bcrypt)
endif()
